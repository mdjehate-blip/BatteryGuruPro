import time
from kivy.app import App
from kivy.uix.boxlayout import BoxLayout
from kivy.uix.gridlayout import GridLayout
from kivy.uix.label import Label
from kivy.uix.button import Button
from kivy.uix.widget import Widget
from kivy.clock import Clock
from kivy.utils import platform
from kivy.graphics import Color, Line, Rectangle, RoundedRectangle

class LiveGraph(Widget):
    def __init__(self, **kwargs):
        super().__init__(**kwargs)
        self.points_data = [0] * 30
        self.bind(pos=self.update_canvas, size=self.update_canvas)

    def add_value(self, val):
        self.points_data.append(abs(val))
        if len(self.points_data) > 30:
            self.points_data.pop(0)
        self.update_canvas()

    def update_canvas(self, *args):
        self.canvas.clear()
        with self.canvas:
            Color(0.12, 0.16, 0.20, 1)
            RoundedRectangle(pos=self.pos, size=self.size, radius=[12])
            
            Color(0.0, 0.78, 0.51, 1)
            max_val = max(max(self.points_data), 500)
            pts = []
            w, h = self.width - 20, self.height - 20
            x0, y0 = self.x + 10, self.y + 10
            
            for i, v in enumerate(self.points_data):
                px = x0 + (i / 29.0) * w
                py = y0 + min((v / float(max_val)), 1.0) * h
                pts.extend([px, py])
                
            if len(pts) >= 4:
                Line(points=pts, width=2.2)

class BatteryDashboard(BoxLayout):
    def __init__(self, **kwargs):
        super().__init__(orientation='vertical', padding=15, spacing=12, **kwargs)
        self.start_time = time.time()
        self.notify_enabled = True

        with self.canvas.before:
            Color(0.07, 0.09, 0.12, 1)
            self.bg_rect = Rectangle(pos=self.pos, size=self.size)
        self.bind(pos=self._update_bg, size=self._update_bg)

        self.header_label = Label(
            text="[b]Battery Guru Pro[/b]", markup=True, font_size='24sp',
            size_hint=(1, 0.1), color=(0.0, 0.85, 0.55, 1)
        )
        self.percent_label = Label(
            text="[b]--%[/b]\nMeasuring...", markup=True, font_size='32sp',
            halign="center", size_hint=(1, 0.2)
        )

        self.grid = GridLayout(cols=2, spacing=10, size_hint=(1, 0.35))
        
        self.card_current = Label(text="Current\n-- mA", markup=True, halign="center", font_size='16sp')
        self.card_temp = Label(text="Temperature\n-- °C", markup=True, halign="center", font_size='16sp')
        self.card_voltage = Label(text="Voltage & Power\n-- mV | -- W", markup=True, halign="center", font_size='16sp')
        self.card_session = Label(text="Screen On\n0s", markup=True, halign="center", font_size='16sp')

        for card in [self.card_current, self.card_temp, self.card_voltage, self.card_session]:
            self.grid.add_widget(card)

        self.graph_title = Label(text="Live Electric Current (mA) Graph", size_hint=(1, 0.05), font_size='14sp', color=(0.7, 0.7, 0.7, 1))
        self.graph = LiveGraph(size_hint=(1, 0.2))

        self.btn_notify = Button(
            text="Live Notification: ON", size_hint=(1, 0.1),
            background_color=(0.0, 0.6, 0.4, 1), bold=True
        )
        self.btn_notify.bind(on_press=self.toggle_notification)

        self.add_widget(self.header_label)
        self.add_widget(self.percent_label)
        self.add_widget(self.grid)
        self.add_widget(self.graph_title)
        self.add_widget(self.graph)
        self.add_widget(self.btn_notify)

        if platform == 'android':
            self.request_android_permissions()

        Clock.schedule_interval(self.update_battery_info, 1.5)

    def _update_bg(self, *args):
        self.bg_rect.pos = self.pos
        self.bg_rect.size = self.size

    def request_android_permissions(self):
        try:
            from jnius import autoclass
            BuildVersion = autoclass('android.os.Build$VERSION')
            if BuildVersion.SDK_INT >= 33:
                PythonActivity = autoclass('org.kivy.android.PythonActivity')
                activity = PythonActivity.mActivity
                activity.requestPermissions(["android.permission.POST_NOTIFICATIONS"], 101)
        except Exception:
            pass

    def toggle_notification(self, instance):
        self.notify_enabled = not self.notify_enabled
        if self.notify_enabled:
            self.btn_notify.text = "Live Notification: ON"
            self.btn_notify.background_color = (0.0, 0.6, 0.4, 1)
        else:
            self.btn_notify.text = "Live Notification: OFF"
            self.btn_notify.background_color = (0.6, 0.2, 0.2, 1)

    def send_live_notification(self, title_str, body_str):
        if not self.notify_enabled or platform != 'android':
            return
        try:
            from jnius import autoclass, cast
            PythonActivity = autoclass('org.kivy.android.PythonActivity')
            Context = autoclass('android.content.Context')
            NotificationManager = autoclass('android.app.NotificationManager')
            NotificationChannel = autoclass('android.app.NotificationChannel')
            NotificationBuilder = autoclass('android.app.Notification$Builder')
            BuildVersion = autoclass('android.os.Build$VERSION')
            String = autoclass('java.lang.String')

            activity = PythonActivity.mActivity
            nm = cast('android.app.NotificationManager', activity.getSystemService(Context.NOTIFICATION_SERVICE))
            channel_id = "battery_live_channel"

            if BuildVersion.SDK_INT >= 26:
                channel = NotificationChannel(channel_id, String("Battery Live Status"), NotificationManager.IMPORTANCE_LOW)
                nm.createNotificationChannel(channel)
                builder = NotificationBuilder(activity, channel_id)
            else:
                builder = NotificationBuilder(activity)

            app_info = activity.getApplicationInfo()
            builder.setContentTitle(String(title_str))
            builder.setContentText(String(body_str))
            builder.setSmallIcon(app_info.icon)
            builder.setOngoing(True)
            nm.notify(1001, builder.build())
        except Exception:
            pass

    def update_battery_info(self, dt):
        elapsed = int(time.time() - self.start_time)
        mins, secs = divmod(elapsed, 60)
        time_str = f"{mins}m {secs}s" if mins > 0 else f"{secs}s"

        if platform == 'android':
            try:
                from jnius import autoclass, cast
                PythonActivity = autoclass('org.kivy.android.PythonActivity')
                Intent = autoclass('android.content.Intent')
                IntentFilter = autoclass('android.content.IntentFilter')
                BatteryManager = autoclass('android.os.BatteryManager')
                Context = autoclass('android.content.Context')

                activity = PythonActivity.mActivity
                ifilter = IntentFilter(Intent.ACTION_BATTERY_CHANGED)
                battery_status = activity.registerReceiver(None, ifilter)

                level = battery_status.getIntExtra(BatteryManager.EXTRA_LEVEL, -1)
                scale = battery_status.getIntExtra(BatteryManager.EXTRA_SCALE, -1)
                percentage = int((level / float(scale)) * 100)

                voltage = battery_status.getIntExtra(BatteryManager.EXTRA_VOLTAGE, -1)
                temp = battery_status.getIntExtra(BatteryManager.EXTRA_TEMPERATURE, -1) / 10.0
                status = battery_status.getIntExtra(BatteryManager.EXTRA_STATUS, -1)
                health = battery_status.getIntExtra(BatteryManager.EXTRA_HEALTH, -1)

                is_charging = status == BatteryManager.BATTERY_STATUS_CHARGING or status == BatteryManager.BATTERY_STATUS_FULL
                state_str = "[color=00e676]Charging ⚡[/color]" if is_charging else "[color=ff9100]Discharging 🔋[/color]"
                health_str = "Good" if health == BatteryManager.BATTERY_HEALTH_GOOD else "Normal"

                bm = cast('android.os.BatteryManager', activity.getSystemService(Context.BATTERY_SERVICE))
                current_now = bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CURRENT_NOW)
                
                current_ma = int(current_now / 1000.0) if abs(current_now) > 10000 else int(current_now)
                if not is_charging and current_ma > 0:
                    current_ma = -current_ma
                elif is_charging and current_ma < 0:
                    current_ma = abs(current_ma)

                power_watt = abs((current_ma / 1000.0) * (voltage / 1000.0))

                self.percent_label.text = f"[b]{percentage}%[/b]\n[size=18sp]{state_str}[/size]"
                self.card_current.text = f"[b]Electric Current[/b]\n[color=00e5ff]{current_ma} mA[/color]"
                self.card_temp.text = f"[b]Temperature[/b]\n[color=ffab40]{temp:.1f} °C[/color] ({health_str})"
                self.card_voltage.text = f"[b]Power & Voltage[/b]\n{power_watt:.1f} W  •  {voltage} mV"
                self.card_session.text = f"[b]Active Session[/b]\nScreen On: {time_str}"

                self.graph.add_value(current_ma)

                notif_title = f"Now: {current_ma} mA ({power_watt:.1f} W) • {temp:.1f}°C"
                notif_body = f"Battery: {percentage}% ({'Charging' if is_charging else 'Discharging'}) • Screen On: {time_str}"
                self.send_live_notification(notif_title, notif_body)

            except Exception as e:
                self.percent_label.text = f"Error: {str(e)}"
        else:
            self.percent_label.text = "[b]39%[/b]\n[size=18sp][color=00e676]Charging ⚡[/color][/size]"
            self.card_current.text = "[b]Electric Current[/b]\n[color=00e5ff]303 mA[/color]"
            self.card_temp.text = "[b]Temperature[/b]\n[color=ffab40]33.8 °C[/color] (Good)"
            self.card_voltage.text = "[b]Power & Voltage[/b]\n1.2 W  •  3830 mV"
            self.card_session.text = f"[b]Active Session[/b]\nScreen On: {time_str}"
            self.graph.add_value(303)

class BatteryApp(App):
    def build(self):
        return BatteryDashboard()

if __name__ == '__main__':
    BatteryApp().run()
